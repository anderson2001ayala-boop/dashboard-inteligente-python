import streamlit as st
import pandas as pd
import plotly.express as px

# CONFIGURACIÓN
st.set_page_config(
    page_title="Mi Primer Dashboard",
    layout="wide"
)

# TÍTULO
st.title("📊 Dashboard Inteligente")
st.sidebar.title("⚙️ Panel de Control")

opcion = st.sidebar.selectbox(
    "Selecciona una opción",
    ["Ventas", "Clientes", "Ganancias"]
)

st.write("Seleccionaste:", opcion)

# DATOS
df = pd.DataFrame({
    "Mes": ["Enero", "Febrero", "Marzo", "Abril"],
    "Ventas": [1200, 1800, 2400, 3000]
})

# MÉTRICAS
col1, col2 = st.columns(2)

col1.metric("Ventas Totales", "$8,400")
col2.metric("Clientes", "230")

# GRÁFICO
fig = px.bar(df, x="Mes", y="Ventas", title="Ventas Mensuales")
st.subheader("📈 Estadísticas")

st.area_chart(df["Ventas"])

st.plotly_chart(fig, use_container_width=True)

# TABLA
st.subheader("Datos")
st.dataframe(df)
